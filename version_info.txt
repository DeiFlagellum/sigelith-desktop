# Zasob VERSIONINFO wkompilowywany w plik .exe (PyInstaller: EXE(version=...)).
# Wersja MUSI zgadzac sie z `__version__` (tests/test_rebrand.py).
#
# Po co: plik bez tego zasobu nie ma w ogole zakladki „Szczegoly" we
# wlasciwosciach Windows — nie widac ani wydawcy, ani wersji, ani opisu.
# Dla uzytkownika to program bez tozsamosci, a dla heurystyk antywirusowych
# i SmartScreena jeden z sygnalow podwyzszajacych ocene ryzyka. Poprzedni
# `tvs_gui.exe` byl budowany bez niego.
#
# To NIE zastepuje podpisu cyfrowego (Authenticode) — metadane moze wpisac
# ktokolwiek. Daje rozpoznawalnosc i porzadek, nie gwarancje pochodzenia.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(3, 0, 2, 0),
    prodvers=(3, 0, 2, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,          # VOS_NT_WINDOWS32
    fileType=0x1,        # VFT_APP
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        # angielski (0x0409) + Unicode (1200): tabela jest jedna, a program
        # mowi w 11 jezykach — Menedzer zadan pokazuje FileDescription
        # kazdemu uzytkownikowi. Wydawca = ten sam co w paczce MSIX.
        '040904b0',
        [StringStruct('CompanyName', 'Adam Koch'),
         StringStruct('FileDescription', 'Sigelith Desktop'),
         StringStruct('FileVersion', '3.0.2.0'),
         StringStruct('InternalName', 'SigelithDesktop'),
         StringStruct('LegalCopyright', '© 2026 Adam Koch · sigelith.org'),
         StringStruct('OriginalFilename', 'SigelithDesktop.exe'),
         StringStruct('ProductName', 'Sigelith Desktop'),
         StringStruct('ProductVersion', '3.0.2'),
         StringStruct('Comments',
                      'Proof that a file existed at a point in time. The file '
                      'never leaves your computer — only its SHA-256 digest goes '
                      'to the public Sigelith log.')])
    ]),
    VarFileInfo([VarStruct('Translation', [0x0409, 1200])])
  ]
)
