# Zasob VERSIONINFO wkompilowywany w plik .exe (PyInstaller: EXE(version=...)).
#
# Po co: plik bez tego zasobu nie ma w ogole zakladki „Szczegoly" we
# wlasciwosciach Windows — nie widac ani wydawcy, ani wersji, ani opisu.
# Dla uzytkownika to program bez tozsamosci, a dla heurystyk antywirusowych
# i SmartScreena jeden z sygnalow podwyzszajacych ocene ryzyka. Poprzedni
# `tvs_gui.exe` byl budowany bez niego.
#
# To NIE zastepuje podpisu cyfrowego (Authenticode) — metadane moze wpisac
# ktokolwiek. Daje rozpoznawalnosc i porzadek, nie gwarancje pochodzenia.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(2, 1, 0, 0),
    prodvers=(2, 1, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,          # VOS_NT_WINDOWS32
    fileType=0x1,        # VFT_APP
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '041504b0',      # polski (0x0415) + Unicode (1200)
        [StringStruct('CompanyName', 'BeatTime'),
         StringStruct('FileDescription', 'BeatStamp — znacznik czasu @beat'),
         StringStruct('FileVersion', '2.1.0.0'),
         StringStruct('InternalName', 'BeatStamp'),
         StringStruct('LegalCopyright', '© BeatTime · beattime.live'),
         StringStruct('OriginalFilename', 'BeatStamp.exe'),
         StringStruct('ProductName', 'BeatStamp'),
         StringStruct('ProductVersion', '2.1.0'),
         StringStruct('Comments',
                      'Dowod istnienia pliku w czasie. Dokument nie opuszcza '
                      'komputera — do rejestru BeatTime trafia wylacznie jego '
                      'skrot SHA-256.')])
    ]),
    VarFileInfo([VarStruct('Translation', [0x0415, 1200])])
  ]
)
